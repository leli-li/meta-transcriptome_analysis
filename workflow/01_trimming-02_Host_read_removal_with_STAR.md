# 01_Trimming
## Trimmomatic installation
# Download Trimmomatic from the official release page.
# The Trimmomatic version used in this workflow is v0.39.
wget -O Trimmomatic-0.39.zip "https://github.com/usadellab/Trimmomatic/files/5854859/Trimmomatic-0.39.zip"
unzip Trimmomatic-0.39.zip

## Trimming
### Create output directory for trimmed reads.
mkdir -p "/path/to/trimmomatic_output"

### Run Trimmomatic
# Replace "sample" with the actual sample identifier and modify the input read paths according to the local sample naming scheme.
# Modify the thread number according to available computing resources.
# The ILLUMINACLIP, SLIDINGWINDOW, and MINLEN settings shown here were used in this workflow and may be adjusted for other datasets if needed.
java -jar "/path/to/Trimmomatic-0.39/trimmomatic-0.39.jar" PE \
  -threads 8 \
  "/path/to/raw_fastq/sample_R1.fastq.gz" \
  "/path/to/raw_fastq/sample_R2.fastq.gz" \
  "/path/to/trimmomatic_output/sample_1P.fq.gz" \
  "/path/to/trimmomatic_output/sample_1U.fq.gz" \
  "/path/to/trimmomatic_output/sample_2P.fq.gz" \
  "/path/to/trimmomatic_output/sample_2U.fq.gz" \
  ILLUMINACLIP:"/path/to/adapter_file.fa":2:30:10 \
  SLIDINGWINDOW:5:20 \
  MINLEN:70
# Only paired reads, corresponding to files marked as *_1P.fq.gz and *_2P.fq.gz, are used in the downstream host mapping step.




# 02_Host_read_removal_with_STAR
# This step uses the paired trimmed reads generated by Trimmomatic.
## STAR installation with Conda
# Conda should be available before running this command.
conda install -c bioconda star=2.7.10a


## Prepare the host reference genome
### Create a directory for the host reference genome
mkdir -p "/path/to/reference_genome"

### Download the host reference genome
# Replace the URL with the download link of the corresponding host genome.
wget -O "/path/to/reference_genome/reference_genome.fa.gz" "url_of_reference_genome" 

### Decompress the reference genome if it is compressed as .gz
gzip -d "/path/to/reference_genome/reference_genome.fa.gz"


## Prepare STAR genome index
### Create a directory for the STAR genome index
mkdir -p "/path/to/star_index"

### Build STAR genome index
# STAR recommends min(14, log2(GenomeLength)/2 - 1) for --genomeSAindexNbases.
# In this workflow, 13 was used for both potato and cassava reference genomes.
# STAR genome indexing requires an uncompressed FASTA file.
STAR \
  --runThreadN 8 \
  --runMode genomeGenerate \
  --genomeDir "/path/to/star_index" \
  --genomeFastaFiles "/path/to/reference_genome/reference_genome.fa" \
  --genomeSAindexNbases 13


## Map trimmed reads to the reference genome
### Create output directory for STAR mapping results
mkdir -p "/path/to/STAR_output"

### Map paired trimmed reads to the host reference genome
# The two files after --readFilesIn correspond to paired read 1 and paired read 2 after Trimmomatic.
# Use --readFilesCommand zcat when input reads are compressed as .gz files.
# Modify the thread number according to available computing resources.
# Replace "sample" with the same sample identifier used in the Trimmomatic output files.
# The STAR output prefix should end with an underscore if output files are expected to begin with "sample_".
# STAR was used for host-read removal rather than host transcript quantification. Therefore, --outSAMtype None was used to avoid writing host-aligned SAM/BAM files.
# --outReadsUnmapped Fastx exports reads that do not map to the host reference genome for downstream microbial profiling.
STAR --runThreadN 8 \
  --genomeDir "/path/to/star_index" \
  --readFilesIn "/path/to/trimmomatic_output/sample_1P.fq.gz" "/path/to/trimmomatic_output/sample_2P.fq.gz" \
  --readFilesCommand zcat \
  --outFileNamePrefix "/path/to/STAR_output/sample_" \
  --outSAMtype None \
  --outReadsUnmapped Fastx
# Main output files used in the downstream plant-read filtering step:
# /path/to/STAR_output/sample*.mate1
# /path/to/STAR_output/sample*.mate2
